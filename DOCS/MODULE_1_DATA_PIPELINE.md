### STEP 2 — Scrape books.toscrape.com

**Goal**: ≥ 60 books across ≥ 3 categories. No login or API key needed.

**Strategy**: Scrape at least 3 category pages (e.g., Mystery, Science, Romance). Each category has paginated listings. Scrape until you have 60+ books total.

**Fields to capture per book**:
| Field | Raw value example |
|---|---|
| `title` | "A Light in the Attic" |
| `price` | "£51.77" (raw string) |
| `star_rating` | "Three" (text class) |
| `availability` | "In stock" (text) |
| `category` | "Mystery" |

**Implementation outline**:

```python
import requests
from bs4 import BeautifulSoup
import pandas as pd

BASE_URL = "https://books.toscrape.com/catalogue/"
CATEGORY_URLS = [
    "https://books.toscrape.com/catalogue/category/books/mystery_3/",
    "https://books.toscrape.com/catalogue/category/books/science_22/",
    "https://books.toscrape.com/catalogue/category/books/romance_8/",
    # Add more if needed to reach 60 books
]

def get_books_from_page(url, category):
    resp = requests.get(url, timeout=10)
    resp.raise_for_status()
    soup = BeautifulSoup(resp.text, "html.parser")
    books = []
    for article in soup.select("article.product_pod"):
        title = article.h3.a["title"]
        price = article.select_one("p.price_color").text.strip()
        rating_word = article.p["class"][1]   # e.g. "Three"
        avail = article.select_one("p.availability").text.strip()
        books.append({
            "title": title,
            "price_raw": price,
            "star_rating": rating_word,
            "availability_raw": avail,
            "category": category
        })
    # Handle pagination
    next_btn = soup.select_one("li.next a")
    next_url = BASE_URL + next_btn["href"] if next_btn else None
    return books, next_url

all_books = []
for cat_url in CATEGORY_URLS:
    url = cat_url
    cat_name = cat_url.split("/")[-2].split("_")[0].title()
    while url:
        books, next_url = get_books_from_page(url, cat_name)
        all_books.extend(books)
        url = next_url

df_raw = pd.DataFrame(all_books)
print(f"Total books scraped: {len(df_raw)}")
print(df_raw["category"].value_counts())
```

**Checkpoint**: `len(df_raw) >= 60` and `df_raw["category"].nunique() >= 3`

### STEP 3 — Clean the Data

**Goal**: Produce properly typed columns: `price_gbp` (float), `rating` (int 1–5), `in_stock` (bool).

```python
# --- price_gbp ---
df_raw["price_gbp"] = df_raw["price_raw"].str.replace(r"[^\d.]", "", regex=True).astype(float)

# --- rating (text → int) ---
rating_map = {"One": 1, "Two": 2, "Three": 3, "Four": 4, "Five": 5}
df_raw["rating"] = df_raw["star_rating"].map(rating_map)

# --- in_stock (bool) ---
df_raw["in_stock"] = df_raw["availability_raw"].str.lower().str.contains("in stock")

# --- Handle parse failures ---
# price_gbp: if NaN, impute with median (justify: median is robust to outliers in price)
price_median = df_raw["price_gbp"].median()
df_raw["price_gbp"] = df_raw["price_gbp"].fillna(price_median)

# rating: if NaN (unexpected text), drop the row (justify: only a handful of rows,
#         ordinal scale makes median imputation unreliable compared to dropping)
df_raw = df_raw.dropna(subset=["rating"])
df_raw["rating"] = df_raw["rating"].astype(int)

# Final clean DataFrame
df_clean = df_raw[["title", "price_gbp", "rating", "in_stock", "category"]].copy()
df_clean = df_clean.reset_index(drop=True)
print(df_clean.dtypes)
print(df_clean.head())
```

**Decisions to document in README**:
- Price NaN → median imputation (median is outlier-robust)
- Rating NaN → drop row (rare; ordinal imputation unreliable)

### STEP 5 — Design & Create the SQLite Schema

**Two-table normalized schema (PK/FK required)**:

```sql
CREATE TABLE IF NOT EXISTS categories (
    category_id   INTEGER PRIMARY KEY AUTOINCREMENT,
    category_name TEXT    UNIQUE NOT NULL
);

CREATE TABLE IF NOT EXISTS books (
    book_id     INTEGER PRIMARY KEY AUTOINCREMENT,
    title       TEXT    NOT NULL,
    price_gbp   REAL    NOT NULL,
    price_inr   REAL    NOT NULL,
    rating      INTEGER NOT NULL,
    in_stock    INTEGER NOT NULL,   -- SQLite has no BOOL; store 0/1
    category_id INTEGER NOT NULL REFERENCES categories(category_id)
);
```

```python
import sqlite3

conn = sqlite3.connect("books.db")
cur = conn.cursor()

cur.execute("DROP TABLE IF EXISTS books")
cur.execute("DROP TABLE IF EXISTS categories")

cur.execute("""
CREATE TABLE categories (
    category_id   INTEGER PRIMARY KEY AUTOINCREMENT,
    category_name TEXT UNIQUE NOT NULL
)
""")

cur.execute("""
CREATE TABLE books (
    book_id     INTEGER PRIMARY KEY AUTOINCREMENT,
    title       TEXT    NOT NULL,
    price_gbp   REAL    NOT NULL,
    price_inr   REAL    NOT NULL,
    rating      INTEGER NOT NULL,
    in_stock    INTEGER NOT NULL,
    category_id INTEGER NOT NULL REFERENCES categories(category_id)
)
""")
conn.commit()
```

### STEP 7 — Write & Execute ≥ 5 SQL Queries

Each query must be saved as a string AND its output printed. Together they must cover:
`SELECT/WHERE`, `ORDER BY`, `LIMIT`, `DISTINCT`, `IN or BETWEEN`, and at least **one JOIN**.

```python
queries = {}

# Q1 — SELECT + WHERE
queries["Q1_cheap_books"] = """
SELECT title, price_gbp, price_inr
FROM books
WHERE price_gbp < 15.00
ORDER BY price_gbp ASC;
"""

# Q2 — ORDER BY + LIMIT
queries["Q2_top10_expensive"] = """
SELECT title, price_gbp, price_inr
FROM books
ORDER BY price_gbp DESC
LIMIT 10;
"""

# Q3 — DISTINCT
queries["Q3_distinct_ratings"] = """
SELECT DISTINCT rating
FROM books
ORDER BY rating;
"""

# Q4 — BETWEEN
queries["Q4_mid_price_range"] = """
SELECT title, price_gbp, rating
FROM books
WHERE price_gbp BETWEEN 20.00 AND 40.00
ORDER BY price_gbp;
"""

# Q5 — IN
queries["Q5_high_ratings"] = """
SELECT title, rating, in_stock
FROM books
WHERE rating IN (4, 5)
ORDER BY rating DESC, title ASC;
"""

# Q6 — JOIN (required: at least one JOIN between the two tables)
queries["Q6_top10_rated_per_category"] = """
SELECT c.category_name, b.title, b.rating, b.price_gbp
FROM books b
JOIN categories c ON b.category_id = c.category_id
ORDER BY c.category_name, b.rating DESC
LIMIT 10;
"""

# Execute and print all
for name, sql in queries.items():
    print(f"\n=== {name} ===")
    print(sql)
    result = pd.read_sql(sql, conn)
    print(result.to_string())
```

### STEP 8 — Read Back with `pd.read_sql` AND `pd.merge`

**Goal**: Show both approaches produce identical output for the JOIN query.

```python
# --- Approach 1: pd.read_sql (SQL JOIN) ---
sql_join = """
SELECT c.category_name, b.title, b.rating, b.price_gbp
FROM books b
JOIN categories c ON b.category_id = c.category_id
ORDER BY c.category_name, b.rating DESC
LIMIT 10;
"""
df_sql = pd.read_sql(sql_join, conn)
print("=== SQL JOIN Result ===")
print(df_sql)

# --- Approach 2: pd.merge (Python-side merge, no SQL) ---
df_books     = pd.read_sql("SELECT * FROM books", conn)
df_categories = pd.read_sql("SELECT * FROM categories", conn)

df_merged = pd.merge(df_books, df_categories, on="category_id")
df_merged = df_merged[["category_name", "title", "rating", "price_gbp"]]
df_merged = df_merged.sort_values(["category_name", "rating"], ascending=[True, False]).head(10)
df_merged = df_merged.reset_index(drop=True)

print("\n=== pd.merge Result ===")
print(df_merged)

# Verify equivalence
pd.testing.assert_frame_equal(df_sql.reset_index(drop=True),
                               df_merged.reset_index(drop=True),
                               check_like=True)
print("\n✅ Both outputs match!")

conn.close()
```
